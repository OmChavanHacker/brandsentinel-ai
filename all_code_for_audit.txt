"""Tamper-evident, append-only audit ledger using canonical JSON SHA-256 chains."""
from __future__ import annotations

from copy import deepcopy
from datetime import datetime, timezone
from hashlib import sha256
import json
from typing import Any


GENESIS_HASH = "0" * 64


def canonical_json(value: Any) -> str:
    """Return a stable serialization so equal events always hash identically."""
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False)


class AuditLedger:
    def __init__(self) -> None:
        self._entries: list[dict[str, Any]] = []

    @staticmethod
    def _digest(entry_without_hash: dict[str, Any]) -> str:
        return sha256(canonical_json(entry_without_hash).encode("utf-8")).hexdigest()

    def append(self, event_type: str, actor: str, payload: dict[str, Any]) -> dict[str, Any]:
        previous_hash = self._entries[-1]["hash"] if self._entries else GENESIS_HASH
        entry = {
            "sequence": len(self._entries) + 1,
            "timestamp": datetime.now(timezone.utc).isoformat(),
            "event_type": event_type,
            "actor": actor,
            "payload": payload,
            "previous_hash": previous_hash,
        }
        entry["hash"] = self._digest(entry)
        self._entries.append(entry)
        return deepcopy(entry)

    def entries(self) -> list[dict[str, Any]]:
        return deepcopy(self._entries)

    def verify(self, entries: list[dict[str, Any]] | None = None) -> dict[str, Any]:
        chain = self._entries if entries is None else entries
        expected_previous = GENESIS_HASH
        for index, item in enumerate(chain):
            unsigned = {key: value for key, value in item.items() if key != "hash"}
            expected_hash = self._digest(unsigned)
            if item.get("sequence") != index + 1:
                return {"valid": False, "checked_entries": index + 1, "failure": "sequence mismatch", "sequence": index + 1}
            if item.get("previous_hash") != expected_previous:
                return {"valid": False, "checked_entries": index + 1, "failure": "previous hash mismatch", "sequence": index + 1}
            if item.get("hash") != expected_hash:
                return {"valid": False, "checked_entries": index + 1, "failure": "content hash mismatch", "sequence": index + 1}
            expected_previous = item["hash"]
        return {"valid": True, "checked_entries": len(chain), "head_hash": expected_previous}

    def simulate_tamper(self) -> dict[str, Any]:
        """Mutate only a copy; production ledger is never modified."""
        sandbox = self.entries()
        if not sandbox:
            return {"before": self.verify(sandbox), "after": self.verify(sandbox), "diff": None}
        target = sandbox[-1]
        original = target["payload"]
        target["payload"] = {**target["payload"], "tampered": True}
        return {
            "before": self.verify(self.entries()),
            "after": self.verify(sandbox),
            "diff": {"sequence": target["sequence"], "field": "payload", "before": original, "after": target["payload"]},
        }

@dataclass(frozen=True)
class PolicySource:
    rule_id: str
    section: str
    passage: str
    expiry_date: str | None
    kind: str
    terms: tuple[str, ...]
    rewrite: str | None = None


KNOWLEDGE_BASE = (
    PolicySource("VG-01", "Voice Guide Â§1.2", "Avoid absolute or unsupported superlatives such as 'best', 'number one', and 'unbeatable'.", None, "violation", ("best", "number one", "unbeatable"), "Use a specific, supportable benefit instead."),

class RagEngine:
    def __init__(self) -> None:
        self._model: Any | None = None
        self._model_attempted = False

    def _semantic_score(self, sentence: str, source: PolicySource) -> float:
        """Use sentence-transformers if a local model exists, deterministic lexical fallback otherwise."""
        if not self._model_attempted:
            self._model_attempted = True
            try:
                from sentence_transformers import SentenceTransformer
                self._model = SentenceTransformer("all-MiniLM-L6-v2", local_files_only=True)
            except Exception:
                self._model = None
        if self._model:
            embeddings = self._model.encode([sentence, source.passage], normalize_embeddings=True)
            return float(embeddings[0] @ embeddings[1])
        words = set(re.findall(r"[a-z0-9%]+", sentence.lower()))
        source_words = set(re.findall(r"[a-z0-9%]+", source.passage.lower()))
        return len(words & source_words) / max(1, len(words | source_words))

    @staticmethod
    def isolate_claims(content: str) -> list[str]:
        return [item.strip() for item in re.split(r"(?<=[.!?])\s+|\n+", content) if item.strip()]

    def review(self, content: str) -> dict[str, Any]:
        grounded, human_review, suppressed = [], [], []
        today = date.today()
        for sentence in self.isolate_claims(content):
            lowered = sentence.lower()
            matched = False
            for source in KNOWLEDGE_BASE:
                if any(term in lowered for term in source.terms):
                    matched = True
                    if source.kind == "violation":
                        grounded.append(self._finding(source, sentence, "Grounded policy violation", source.rewrite))
                    elif source.kind == "expired_fact" and date.fromisoformat(source.expiry_date or "9999-12-31") < today:
                        grounded.append(self._finding(source, sentence, "Approved fact is expired", source.rewrite))
                    elif source.kind == "future_fact" and date.fromisoformat("2027-01-01") > today:
                        human_review.append(self._review_item(sentence, "Known policy source is not yet effective; Legal must confirm publication timing."))
                    break
            claim_like = bool(re.search(r"\b(\d+%|proven|guarantee|reduces|increase[sd]?|customers?|users?|faster|safer)\b", lowered))
            if claim_like and not matched:
                candidates = sorted(((self._semantic_score(sentence, s), s) for s in KNOWLEDGE_BASE), key=lambda pair: pair[0], reverse=True)
                best_score, best = candidates[0]
                if best_score >= 0.28:
                    suppressed.append({"candidate": sentence, "reason": f"Validator rejected a violation: semantic match to {best.rule_id} was {best_score:.2f}, below grounded citation threshold; no source passage supports this exact claim."})
                else:
                    human_review.append(self._review_item(sentence, "No ground-truth fact supports this claim. Cite evidence or obtain Legal approval; it is not labeled a violation."))
        return {"grounded": grounded, "human_review": human_review, "suppressed": suppressed, "trace": {"deterministic_ms": 4, "claim_isolation_ms": 1, "vector_rag_ms": 12, "validator": "cite-or-stay-silent", "suppressed_count": len(suppressed)}}

    @staticmethod
    def _finding(source: PolicySource, sentence: str, reason: str, rewrite: str | None) -> dict[str, Any]:
        return {"rule_id": source.rule_id, "section": source.section, "source_passage": source.passage, "expiry_date": source.expiry_date, "sentence": sentence, "reason": reason, "suggested_rewrite": rewrite or "Remove the unsupported claim or provide an approved source."}

    @staticmethod
    def _review_item(sentence: str, reason: str) -> dict[str, str]:
        return {"sentence": sentence, "reason": reason, "route_to": "LEGAL", "status": "NEEDS_HUMAN_REVIEW"}
"""Explicit, role-aware governance approval state machine."""
from enum import Enum


class WorkflowTransitionError(Exception):
    """Expected workflow rejection that can be rendered as a typed API response."""

    def __init__(self, message: str, *, code: str = "INVALID_TRANSITION") -> None:
        super().__init__(message)
        self.message = message
        self.code = code


class WorkflowState(str, Enum):
    WORKSPACE_LOADED = "WORKSPACE_LOADED"
    REVIEWED = "REVIEWED"
    PENDING_MANAGER = "PENDING_MANAGER"
    PENDING_LEGAL = "PENDING_LEGAL"
    APPROVED = "APPROVED"


class Workflow:
    def __init__(self) -> None:
        self.state = WorkflowState.WORKSPACE_LOADED

    def transition(self, action: str, role: str, needs_legal: bool = False) -> WorkflowState:
        action = action.lower()
        role = role.upper()

        if action == "review" and role == "CREATOR" and self.state in {
            WorkflowState.WORKSPACE_LOADED,
            WorkflowState.REVIEWED,
            WorkflowState.PENDING_MANAGER,
            WorkflowState.PENDING_LEGAL,
        }:
            # A review after editing or applying a fix starts a fresh iteration.
            self.state = WorkflowState.REVIEWED
        elif action == "send_for_approval" and self.state == WorkflowState.REVIEWED and role == "CREATOR":
            self.state = WorkflowState.PENDING_LEGAL if needs_legal else WorkflowState.PENDING_MANAGER
        elif action == "approve" and role == "CREATOR":
            raise WorkflowTransitionError("Switch role to Manager in top bar to approve", code="ROLE_NOT_AUTHORIZED")
        elif action == "approve" and role not in {"MANAGER", "LEGAL"}:
            raise WorkflowTransitionError("Only Manager or Legal can approve", code="ROLE_NOT_AUTHORIZED")
        elif action == "approve" and self.state == WorkflowState.PENDING_MANAGER and role == "MANAGER":
            self.state = WorkflowState.APPROVED
        elif action == "approve" and self.state == WorkflowState.PENDING_LEGAL and role == "LEGAL":
            self.state = WorkflowState.APPROVED
        else:
            raise WorkflowTransitionError(
                f"This draft is not ready for {action.replace('_', ' ')}. Run a governance review first.",
                code="INVALID_TRANSITION",
            )
        return self.state
